import sys,json,base64
from pathlib import Path
from copy import deepcopy
ROOT=Path(r'C:/Users/laichungling/Documents/ChatGPT/pdf-to-md-agent-2')
sys.path[:0]=[str(ROOT/'src'),str(ROOT/'tests')]
import pymupdf,httpx
from pdfmd.evidence import Evidence
from pdfmd.storage import Store
from pdfmd.runtime import Runtime
from pdfmd.nexus import Nexus
from test_context_retry import pressure,POLICY
from test_context_admission import state_for
from test_stencil_pins import accepted,offline_runtime
from test_runtime import tool
from test_response_repairs import native,handoff
T=Path(__file__).parent
results=[]
for protocol in ('chat_completions','responses'):
 for scenario in ('omitted_notes','clear_notes','replace_notes','release_repin'):
  d=T/(protocol+'-'+scenario);d.mkdir()
  with pymupdf.open() as pdf:
   pdf.new_page().insert_text((72,72),'Keep 25 mg.');pdf.save(d/'source.pdf')
  store=Store(d/'workspace');evidence=Evidence(store)
  job=evidence.create_job(str(d/'source.pdf'),'Convert the complete page.',{'endpoint':'http://127.0.0.1:8317/v1/chat/completions','model':'offline-fixture','context_tokens':65536,'output_tokens':2048})
  job,state,_,ref=pressure(store,job,protocol)
  before=deepcopy(state)
  first={'notes':'Keep both 25 mg and the complete source.'}
  second={'pin':[{'ref':ref,'view':'index'}]}
  expected=first['notes']
  if scenario=='clear_notes': second['notes']='';expected=''
  if scenario=='replace_notes': second['notes']='Replacement notes.';expected=second['notes']
  if scenario=='release_repin':
   second={'release':[ref],'pin':[{'ref':ref,'view':'full','json_pointer':'/text','start':10,'end':30}]}
  huge=store.blob(job['id'],'image_stream',b'X'*65536)['ref']
  # Genuine body read, two valid updates, then two invalid updates in one accepted batch.
  accepted(store,job,'pending',tool('read_evidence',{'ref':ref},'read'),tool('context_update',first,'first'),
   tool('context_update',second,'second'),tool('context_update',{'notes':None},'invalid-null'),
   tool('context_update',{'pin':[{'ref':huge,'view':'raw'}],'notes':'MUST NOT APPLY'},'invalid-raw'))
  rt=offline_runtime(store)
  state.update(rt.tools(state))
  pending=state['context_waiting'];assert pending=='result:pending:2'
  proposal=rt.context.waiting(job['id'],state);policy=proposal['policy']
  assert policy['notes']==expected and ref in policy['pinned'] and ref not in policy['released']
  assert policy['pin_views'][ref]==second['pin']
  assert all(state[k]==before.get(k,{} if k=='pin_views' else '') for k in POLICY)
  assert not store.get('operation','pending:3')['ok'] and not store.get('operation','pending:4')['ok']
  original=store.bytes(pending,job['id']);config={'configurable':{'thread_id':job['id']}}
  rt.graph.update_state(config,state,as_node='tools');rt.close()
  store.update_job(job['id'],attempts=7,active_seconds=17)
  seen=[]
  def respond(request):
   payload=json.loads(request.content);seen.append(payload)
   assert len(seen)<=8
   if not payload.get('tools'):
    return httpx.Response(200,json=native(protocol,text=json.dumps(handoff(job,current_state='Source is not complete.',unfinished_work=['Continue conversion.'],references=[ref],cautions=[]))))
   visible=json.dumps(payload).replace('\\"','"')
   assert '"status": "applied"' in visible
   assert pending in visible
   return httpx.Response(200,json=native(protocol,tool('report',{'state':'waiting','message':'Local audit complete.'},'done')))
  reopened=Store(store.root);rt=Runtime(reopened,Nexus(reopened,lambda _:'offline',httpx.MockTransport(respond)))
  rt.run(job['id']);final=dict(rt.graph.get_state(config).values);rt.close()
  current=reopened.get('job',job['id']);assert current['stop_reason']=='agent_report',current
  assert all(final[k]==policy[k] for k in POLICY)
  assert current['attempts']==7+len(seen) and current['active_seconds']>=17
  assert reopened.bytes(pending,job['id'])==original
  assert reopened.json(pending,job['id'])['admission']['applied'] is False
  assert base64.b64decode(Evidence(reopened).read(job['id'],ref,view='raw')['data'])==store.bytes(ref,job['id'])
  assert Evidence(reopened).read(job['id'],ref,json_pointer='/text',start=10,end=30)['value']=='A'*20
  events=[e for e in reopened.events(job['id']) if e['type']=='context_update_settled']
  assert len(events)==1
  results.append({'protocol':protocol,'scenario':scenario,'notes':expected,'unchanged_capacity':job['route']['context_tokens'],
   'pending_ref':pending,'settlements':len(events),'mock_requests':len(seen),'compactions':sum(not p.get('tools') for p in seen),
   'original_receipt_unchanged':True,'invalid_followups_preserve_valid_work':True,'raw_and_range_exact':True,
   'consumption_retained':True,'continued_after_reopen':True,'live_calls':0})
(T/'adjacent-results.json').write_text(json.dumps(results,indent=2),encoding='utf8')
print(json.dumps(results,indent=2))
