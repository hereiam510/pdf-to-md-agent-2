def warnings(value, path=""):
    """Keep diagnostic fields by structure, without judging source meaning."""
    found = {}
    if isinstance(value, dict):
        for key, item in value.items():
            child = path + "/" + key.replace("~", "~0").replace("/", "~1")
            if key in {"warnings", "uncertainties", "limitations"}:
                found[child] = item
            else:
                found.update(warnings(item, child))
    elif isinstance(value, list):
        for i, item in enumerate(value):
            found.update(warnings(item, f"{path}/{i}"))
    return found
