    async def retry_wait(self, audit):
        """Resume a persisted cancellable wait without spending another attempt."""
        self.status["phase"] = (
            "queue_retry_wait"
            if audit.get("retry_reason") == "inference_queue_timeout"
            else "automatic_retry_wait"
        )
        self.publish()
        remaining = max(
            0,
            (datetime.fromisoformat(audit["retry_not_before"]) - datetime.now(UTC)).total_seconds(),
        )
        started = time.monotonic()
        try:
            await anyio.sleep(remaining)
        finally:
            audit["retry_wait_elapsed_seconds"] = round(
                audit.get("retry_wait_elapsed_seconds", 0) + time.monotonic() - started, 3
            )
        audit["state"] = (
            "queue_timeout_retried"
            if audit.get("retry_reason") == "inference_queue_timeout"
            else "transient_retried"
        )
        self.status["phase"] = "model"
        self.publish()
