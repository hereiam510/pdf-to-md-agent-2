def pointer(value, path: str):
    if path and not path.startswith("/"):
        raise ValueError("Use an RFC 6901 JSON pointer, or empty string for the full record")
    for key in path.split("/")[1:]:
        if re.search(r"~(?:[^01]|$)", key):
            raise ValueError("Invalid JSON pointer escape")
        key = key.replace("~1", "/").replace("~0", "~")
        if isinstance(value, list):
            if not re.fullmatch(r"0|[1-9][0-9]*", key):
                raise ValueError("Array JSON pointers require a nonnegative decimal index")
            value = value[int(key)]
        else:
            value = value[key]
    return value


def describe_value(value, path: str) -> dict:
    """Describe exact structure and sizes without copying source values into model input."""

    def shape(item, location):
        result = {"pointer": location}
        if isinstance(item, dict):
            result.update(type="object", fields=list(item))
        elif isinstance(item, list):
            result.update(type="array", items=len(item))
        elif isinstance(item, str):
            result.update(type="string", characters=len(item), utf8_bytes=len(item.encode()))
        elif item is None:
            result["type"] = "null"
        elif isinstance(item, bool):
            result["type"] = "boolean"
        else:
            result["type"] = "number"
        return result

    result = shape(value, path)
    if isinstance(value, dict):
        result["fields"] = {
            key: shape(item, path + "/" + key.replace("~", "~0").replace("/", "~1"))
            for key, item in value.items()
        }
    return result

    def read(
        self,
        evidence: str | None,
        pointer_path="",
        offset=None,
        limit=None,
        arguments=False,
        describe=False,
    ):
        if evidence is None:
            if arguments:
                raise ValueError("arguments requires an evidence ID from the index")
            raw = self.records.copy()
        else:
            try:
                suffix = "-arguments" if arguments else ""
                raw = read_json(self.run / f"{identifier(evidence)}{suffix}.json")
            except FileNotFoundError:
                raise ValueError(f"Unknown evidence ID: {evidence}") from None
        value = pointer(raw, pointer_path)
        if offset is None and limit is None:
            result = {
                "evidence": evidence,
                "pointer": pointer_path,
                "complete": True,
                "value": value,
            }
        else:
            if not isinstance(value, (list, str)):
                raise ValueError("offset/limit apply only to arrays or strings")
            start = offset or 0
            if start < 0 or start > len(value) or (limit is not None and limit < 1):
                raise ValueError("Invalid evidence slice")
            end = min(len(value), start + limit) if limit is not None else len(value)
            result = {
                "evidence": evidence,
                "pointer": pointer_path,
                "total": len(value),
                "offset": start,
                "end": end,
                "next_offset": end if end < len(value) else None,
                "complete": start == 0 and end == len(value),
                "value": value[start:end],
            }
        if describe:
            result["described"] = True
            result["metadata"] = describe_value(result.pop("value"), pointer_path)
        return result
