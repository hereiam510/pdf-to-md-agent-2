class Model(BaseModel):
    model_config = ConfigDict(extra="forbid")


class Page(Model):
    pdf_page: int = Field(ge=1)
    printed_page: str = "unknown"
    disposition: Literal["in_scope", "outside_scope", "unprocessed"] = "unprocessed"


class Block(Model):
    id: str = Field(pattern=r"^[a-zA-Z0-9_-]+$")
    pages: list[int] = Field(min_length=1)
    markdown: str = Field(min_length=1)


class Content(Model):
    title: str = Field(min_length=1)
    language: str = Field(default="und", pattern=r"^[A-Za-z]{2,8}(-[A-Za-z0-9]{1,8})*$")
    scope: str = Field(min_length=1)
    status: Literal["draft", "complete"] = "draft"
    pages: list[Page]
    blocks: list[Block] = Field(min_length=1)
    uncertainties: list[str] = Field(default_factory=list)
    limitations: list[str] = Field(default_factory=list)
    metrics: dict[str, str | int | float | None] = Field(default_factory=dict)
