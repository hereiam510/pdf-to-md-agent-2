    async def recognize(
        self,
        document: str,
        pages: list[int],
        languages: list[str] | None = None,
        dpi: int = 300,
        region: list[float] | None = None,
        psm: int = 3,
        timeout: float = 120,
    ) -> dict[str, Any]:
        """Recognize fresh page/region images with local language data, positions and warnings.

        languages defaults to chi_tra + eng. Coordinates match inspect/render_page.
        Select PSM 3 (auto), 6 (block), 7 (line), 11 (sparse), etc.; OSD requires extra data.
        No source/delivered graphics are changed. Inspect OCR and visible source before composing.
        """
        languages = ["chi_tra", "eng"] if languages is None else languages
        if (
            not languages
            or len(set(languages)) != len(languages)
            or any(not re.fullmatch(r"[a-z0-9_]+", language) for language in languages)
        ):
            raise ValueError("Select distinct OCR language codes, such as ['chi_tra', 'eng']")
        if not pages or len(set(pages)) != len(pages):
            raise ValueError("Select one or more distinct physical pages")
        if (
            not isinstance(psm, int)
            or isinstance(psm, bool)
            or psm not in {3, 4, 5, 6, 7, 8, 9, 10, 11, 13}
        ):
            raise ValueError("Use a recognition PSM in 3..11 or 13; OSD is not installed")
        if not 0 < timeout <= 300:
            raise ValueError("timeout must be in (0, 300] seconds per page")
