    def export(self, document: str, version: str | None = None) -> dict[str, Any]:
        saved = self.reopen(document, version)
        version = saved["version"]
        folder = self.task(document) / "versions" / version
        manifest = read_json(folder / "manifest.json")
        outputs = self.task(document) / "exports"
        outputs.mkdir(exist_ok=True)
        destination = outputs / f"{version}.zip"
        temporary = outputs / f".{uuid.uuid4().hex}.zip"
        try:
            with zipfile.ZipFile(temporary, "w", compression=zipfile.ZIP_DEFLATED) as archive:
                for name in ["manifest.json", *manifest["files"]]:
                    archive.write(folder / name, name)
