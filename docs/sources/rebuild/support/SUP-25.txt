class NativeObserver:
    """Observe complete LF/CRLF Chat SSE frames without emitting model content.

    Only the current frame is inspected; forwarded bytes are never reconstructed.
    No partial argument is executed here.
    """

    def __init__(self):
        self.line = bytearray()
        self.fields = []
        self.frames = 0
        self.meaningful_frames = 0
        self.substantive_characters = 0
        self.terminal = False
        self.first_meaningful = None
        self.last_meaningful = None
        self.terminal_at = None

    def feed(self, data, *, interrupt=False):
        start = 0
        while start < len(data):
            end = data.find(b"\n", start)
            stop = len(data) if end < 0 else end + 1
            part = data[start:stop]
            self.line.extend(part)
            if end < 0:
                return None
            line = bytes(self.line).rstrip(b"\r\n")
            self.line.clear()
            if not line:
                cut = self._frame(interrupt)
                if cut:
                    return stop
            elif line.startswith(b"data:"):
                self.fields.append(line[5:].removeprefix(b" "))
            start = stop
        return None

    def _frame(self, interrupt):
        fields, self.fields = self.fields, []
        if not fields:
            return False
        data = b"\n".join(fields)
        self.frames += 1
        characters, terminal = 0, data == b"[DONE]"
        try:
            event = json.loads(data)
            if (
                isinstance(event, dict)
                and event.get("object") == "chat.completion.chunk"
                and isinstance(event.get("id"), str)
                and isinstance(event.get("choices"), list)
            ):
                for choice in event["choices"]:
                    terminal |= choice.get("finish_reason") is not None
                    delta = choice.get("delta", {})
                    text = delta.get("content")
                    if isinstance(text, str):
                        characters += sum(c.isalnum() for c in text)
                    for call in delta.get("tool_calls", []):
                        arguments = call.get("function", {}).get("arguments")
                        if isinstance(arguments, str):
                            characters += sum(c.isalnum() for c in arguments)
        except (ValueError, TypeError, AttributeError):
            characters = 0
        now = time.monotonic()
        if terminal:
            self.terminal, self.terminal_at = True, now
        if characters:
            self.substantive_characters += characters
            self.meaningful_frames += 1
            self.first_meaningful = self.first_meaningful or now
            self.last_meaningful = now
        return (
            interrupt
            and characters > 0
            and self.substantive_characters >= MEANINGFUL_CHARACTERS
            and not self.terminal
        )

    async def close(self):
        self.server.close()
        if self.gate_timer:
            if not self.gate_timer.done():
                self.event("gate_stopped", scheduled_end_monotonic=self.gate_end)
            self.gate_timer.cancel()
            await asyncio.gather(self.gate_timer, return_exceptions=True)
        for task in list(self.tasks):
            task.cancel()
        await asyncio.gather(*self.tasks, return_exceptions=True)
        for writer in list(self.handles):
            await self.close_writer(writer)
        await self.server.wait_closed()
        if not self.injected:
            self.event(
                "fault_unexercised",
                reason=(
                    "no_acknowledged_partial_window"
                    if self.armed is None
                    else "no_qualifying_native_window_before_stop"
                ),
            )
        self.event(
            "stopped",
            armed=self.armed is not None,
            injected=self.injected,
            open_handles=len(self.handles),
        )

    async def serve(self):
        await self.start()
        try:
            while not self.stop_file.exists():
                await asyncio.sleep(0.2)
        finally:
            await self.close()
