    @staticmethod
    def page(pdf, number: int):
        if isinstance(number, bool) or not isinstance(number, int) or not 1 <= number <= len(pdf):
            raise ValueError(f"PDF page must be an integer in 1..{len(pdf)}")
        return pdf[number - 1]

    @staticmethod
    def region(page, region: list[float] | None):
        # All public coordinates use unrotated PDF points, top left origin.
        bounds = page.rect * page.derotation_matrix
        rect = pymupdf.Rect(region) if region is not None else bounds
        if not all(math.isfinite(v) for v in rect) or rect.is_empty or not bounds.contains(rect):
            raise ValueError(f"Region must be a finite non-empty rectangle inside {list(bounds)}")
        return rect

    def render(
        self,
        document: str,
        page: int,
        dpi: int = 150,
        region: list[float] | None = None,
        asset: bool = False,
    ) -> dict[str, Any]:
        if isinstance(dpi, bool) or not isinstance(dpi, int) or dpi <= 0:
            raise ValueError("dpi must be a positive integer")
        task = self.task(document)
        with self.source(document) as pdf:
            p = self.page(pdf, page)
            rect = self.region(p, region)
            pix = p.get_pixmap(dpi=dpi, clip=rect * p.rotation_matrix, alpha=False)
            data = pix.tobytes("png")
            info = {
                "kind": "page_render",
                "pdf_page": page,
                "region": list(rect),
                "dpi": dpi,
                "width": pix.width,
                "height": pix.height,
                "source_sha256": read_json(task / "source.json")["sha256"],
            }
        if asset:
            return self.store_asset(task, data, "png", info)
        path = task / "evidence" / f"{uuid.uuid4().hex}.png"
        path.write_bytes(data)
        return {**info, "path": str(path)}

    def extract_image(self, document: str, page: int, xref: int) -> dict[str, Any]:
        task = self.task(document)
        with self.source(document) as pdf:
            p = self.page(pdf, page)
            displayed = [im for im in p.get_image_info(xrefs=True) if im["xref"] == xref]
            if not displayed:
                raise ValueError("Image xref is not displayed on the requested page")
            info = pdf.extract_image(xref)
            if not info:
                raise ValueError("Image cannot be extracted; use a faithful page-region render")
            if (
                info.get("smask")
                or any(im.get("has-mask") for im in displayed)
                or pdf.xref_get_key(xref, "ImageMask")[1] == "true"
            ):
                raise ValueError("Masked image requires composition; use a faithful region render")
            meta = {
                "kind": "embedded_image",
                "pdf_page": page,
                "xref": xref,
                "width": info["width"],
                "height": info["height"],
                "source_sha256": read_json(task / "source.json")["sha256"],
                "encoding": info["ext"],
                "extraction": "Document.extract_image",
            }
            return self.store_asset(task, info["image"], info["ext"], meta)
