    def payload(self, history: list[dict], tools: list[dict]) -> dict:
        body = {
            "model": self.config.model,
            "stream": self.config.transport == "streaming",
            **self.config.parameters,
        }
        if self.config.protocol == "chat_completions":
            if body["stream"]:
                options = body.get("stream_options", {})
                if not isinstance(options, dict):
                    raise ValueError("stream_options must be an object")
                body["stream_options"] = {**options, "include_usage": True}
            body.update(
                messages=history, tools=[{"type": "function", "function": t} for t in tools]
            )
        else:
            body.setdefault("include", ["reasoning.encrypted_content"])
            body.update(
                input=history,
                store=False,
                tools=[{"type": "function", **t} for t in tools],
            )
