def replace_file(source: Path, destination: Path):
    # Ordinary Windows readers and scanners may briefly deny DELETE sharing. Retry only
    # those native errors, with a finite budget; permanent failures must still surface.
    for attempt in range(8):
        try:
            _replace_once(source, destination)
            return
        except OSError as error:
            # ERROR_UNABLE_TO_REMOVE_REPLACED (1175) also leaves both original names intact.
            # Unlike 1176/1177, retrying it cannot adopt an indeterminate replacement.
            if os.name != "nt" or error.winerror not in {5, 32, 33, 1175} or attempt == 7:
                raise
            time.sleep(min(0.01 * 2**attempt, 0.16))


def write_json(path: Path, data):
    temporary = path.with_name(f".{path.name}-{uuid.uuid4().hex}.tmp")
    try:
        with temporary.open("w", encoding="utf-8", newline="\n") as stream:
            stream.write(json.dumps(data, ensure_ascii=False, indent=2) + "\n")
            stream.flush()
            os.fsync(stream.fileno())
        replace_file(temporary, path)
    finally:
        temporary.unlink(missing_ok=True)


def shared_read(path: Path):
    """Open a stable old-or-new snapshot, allowing Windows atomic rename while reading."""
    if os.name != "nt":
        return path.open("rb")
    # ReplaceFileW can expose a brief name-transition window to a new opener.
    # Never return partial JSON; only transient Windows open errors are retried.
    for attempt in range(8):
        try:
            return _windows_read(path)
        except OSError as error:
            if error.winerror not in {2, 5, 32, 33} or attempt == 7:
                raise
            time.sleep(min(0.01 * 2**attempt, 0.16))
