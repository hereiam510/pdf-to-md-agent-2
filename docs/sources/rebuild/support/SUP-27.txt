async def inspect_preview(
    workspace: Workspace,
    document: str,
    version: str | None = None,
    width: int = 1200,
    height: int = 900,
    scroll_y: int = 0,
    full_page: bool = False,
) -> dict:
    for name, value, minimum in [
        ("width", width, 1),
        ("height", height, 1),
        ("scroll_y", scroll_y, 0),
    ]:
        if isinstance(value, bool) or not isinstance(value, int) or value < minimum:
            raise ValueError(f"{name} must be an integer >= {minimum}")
    if full_page and scroll_y:
        raise ValueError("Use scroll_y for a viewport, or full_page for the entire reading")
    saved = workspace.reopen(document, version)
    folder = workspace.task(document) / "versions" / saved["version"]
    manifest = read_json(folder / "manifest.json")
    # Freeze verified bytes, not mutable task assets. No temporary HTML rewrite.
    files = {name: (folder / name).read_bytes() for name in manifest["files"]}
    if any(digest(data) != manifest["files"][name] for name, data in files.items()):
        raise ValueError("Saved artifact changed before preview inspection")
    run = workspace.task(document) / "evidence" / f"preview-{uuid.uuid4().hex}"
    run.mkdir()
    status = {
        "document": document,
        "version": saved["version"],
        "preview_sha256": manifest["files"]["preview.html"],
        "status": "running",
        "viewport": {"width": width, "height": height},
        "requested_scroll_y": scroll_y,
        "full_page": full_page,
